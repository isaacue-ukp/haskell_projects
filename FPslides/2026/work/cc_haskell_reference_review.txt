foldr :: (a -> b -> b) -> b -> [a] -> b
foldl :: (b -> a -> b) -> b -> [a] -> b

data Tree a = Leaf a | Node (Tree a) (Tree a) deriving (Eq,Show)
instance Foldable Tree where
  foldMap f (Leaf x) = f x
  foldMap f (Node l r) = foldMap f l <> foldMap f r
instance Traversable Tree where
  traverse f (Leaf x) = Leaf <$> f x
  traverse f (Node l r) = Node <$> traverse f l <*> traverse f r

Parser assumes 0 or 1 result (per lecture):
newtype Parser a = P (String -> [(a,String)])
app (P f) = f
pure x = P $ \s -> [(x,s)]
p >>= k = P $ \s -> case app p s of { [] -> []; [(x,out)] -> app (k x) out }
p <|> q = P $ \s -> case app p s of { [] -> app q s; rs -> rs }
empty = P $ \_ -> []
item = P $ \s -> case s of { [] -> []; (x:xs) -> [(x,xs)] }
sat p = do { x <- item; if p x then pure x else empty }
char c = sat (== c)
many p = some p <|> pure []
some p = (:) <$> p <*> many p
nat = read <$> some (sat isDigit)
space = many (sat isSpace) >> pure ()
token p = do { space; x <- p; space; pure x }
symbol s = token (string s)
string [] = pure []
string (x:xs) = (:) <$> char x <*> string xs
expr = do { t <- term;
  (do { symbol "+"; e <- expr; pure (t+e) }) <|> pure t }
term = do { f <- factor;
  (do { symbol "*"; t <- term; pure (f*t) }) <|> pure f }
factor = (do { symbol "("; e <- expr; symbol ")"; pure e }) <|> token nat
Here right recursion yields right associativity; * binds tighter than +. Check remaining input equals empty for complete parsing. many may diverge if parser succeeds without consuming input.

Monoid requires Semigroup; <> associative, mempty two-sided identity. []: (++),[]; Sum: (+),0; Product: (*),1; All: (&&),True; Any: (||),False.
fold :: (Foldable t, Monoid a) => t a -> a
foldMap :: (Foldable t, Monoid b) => (a -> b) -> t a -> b
fold = foldMap id
Foldable minimal: foldMap or foldr.
